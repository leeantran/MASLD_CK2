df_clean <- df %>%
  mutate(
    # 1. Phân độ gan nhiễm mỡ (Steatosis Grades)
    muc_do_nhiem_mo = case_when(
      cap < 248 ~ "S0",
      cap >= 248 & cap <= 267 ~ "S1",
      cap >= 268 & cap <= 279 ~ "S2",
      cap >= 280 ~ "S3",
      TRUE ~ NA_character_
    ),
    
    # 2. Phân độ xơ hóa (Fibrosis Grades - Tạm dùng cut-off FibroScan)
    muc_do_xo_hoa = case_when(
      fibro < 7.0 ~ "F0-F1",
      fibro >= 7.0 & fibro < 9.5 ~ "F2",
      fibro >= 9.5 & fibro < 12.5 ~ "F3",
      fibro >= 12.5 ~ "F4",
      TRUE ~ NA_character_
    ),
    
    # 3. Yếu tố RLCH (Biến 0/1)
    co_rlch = if_else(
      vong_eo_90_m_80_f == 1 |
      glucose_100_mg_dl == 1 |
      hba1c_5_7_percent == 1 | # Tùy clean_names() ra chữ percent hay không
      dtd2 == 1 |
      ha_130_85 == 1 |
      tha == 1 |
      triglyceride_150_5mg_dl == 1 |
      hdl_40_m_50_f == 1 |
      thuoc_lipid == 1,
      1, 0
    ),
    
    # 4. Chẩn đoán BGNMLQRLCH (MASLD)
    bgnmlqrlch_chuan_doan = if_else(cap >= 248 & co_rlch == 1, "Có", "Không"),
    bgnmlqrlch_binary = if_else(cap >= 248 & co_rlch == 1, 1, 0)
  )
# Bảng 1: Chỉ số TyG-WC và FLI theo chẩn đoán BGNMLQRLCH
